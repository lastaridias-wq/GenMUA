async function verifyProCode() { const inputCode = document.getElementById('proCodeInput').value.trim().toUpperCase(); if (!inputCode) { showToast("⚠️ Masukkan kode aktivasi terlebih dahulu!"); return; } // Trik khusus jika cache limit membandel (Ketik kata RESET di kolom kode) if (inputCode === "RESET") { localStorage.removeItem('genmua_deviceCreatedCount'); appState.projectCount = 0; showToast("🔄 Cache Limit Direset! Silakan tutup pop-up dan coba lagi."); closeModal('upgradeModal'); return; } showToast("🔍 Memeriksa kode ke database..."); try { // Menghubungi Google Apps Script untuk mengecek kode di Data_Lisensi const response = await fetch(`${scriptURL}?action=verifyCode&code=${encodeURIComponent(inputCode)}`); const result = await response.json(); if (result.success) { // 1. Simpan status PRO localStorage.setItem('genmua_isPro', 'true'); // 2. HAPUS CACHE LIMIT YANG MEMBANDEL localStorage.removeItem('genmua_deviceCreatedCount'); activateProUI(); closeModal('upgradeModal'); showToast("🚀 Berhasil! Akun Anda kini berstatus PRO."); } else { // Menggunakan showToast agar TIDAK ADA pop-up github.io showToast("❌ Kode salah, kedaluwarsa, atau tidak terdaftar!"); } } catch (error) { showToast("❌ Gagal terhubung ke database. Periksa koneksi internet."); } }
